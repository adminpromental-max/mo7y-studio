# Portfolio media folders
# ضع الصور والفيديوهات هنا ثم حدّث src/data/portfolio.ts

photos/
  restaurants/   ← صور المطاعم
  cafes/         ← صور الكافيهات
  events/        ← الفعاليات
  products/      ← المنتجات

videos/
  thumbs/        ← صور مصغرة للفيديو
  vertical/      ← ملفات mp4 طولية (اختياري)
  horizontal/    ← ملفات mp4 عرضية (اختياري)

# مثال مسار صورة في portfolio.ts:
# image: "/portfolio/photos/restaurants/burger-01.jpg"

# مثال فيديو محلي:
# image: "/portfolio/videos/thumbs/reel-01.jpg"
# videoUrl: "/portfolio/videos/vertical/reel-01.mp4"
